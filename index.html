JavaScript Frontend Integration 
import {
    createAuthConfirm
} from "./src/auth-confirm.js";

const auth = createAuthConfirm({

    theme: "auto",

    primaryColor: "#078b4d",

    onConfirm: async (action, { signal }) => {

        const response = await fetch(
            `/api/${action}.php`,
            {
                method: "POST",
                signal
            }
        );

        return response.ok;
    }

});

await auth.confirmLogin();
PHP Backend Integration 
<?php

session_start();

header("Content-Type: application/json");

if ($_SERVER["REQUEST_METHOD"] !== "POST") {
    http_response_code(405);
    exit;
}

$action = basename(
    $_SERVER["SCRIPT_NAME"],
    ".php"
);

if ($action === "login") {

    // Validate user credentials here.

    $_SESSION["user"] = [
        "id" => 1,
        "name" => "John Doe"
    ];

} elseif ($action === "logout") {

    session_unset();
    session_destroy();

} else {

    http_response_code(400);

    echo json_encode([
        "success" => false
    ]);

    exit;
}

echo json_encode([
    "success" => true
]);

Custom Login Confirmation

await auth.confirmLogin({

    title: "Welcome Back!",

    message: "Ready to continue?",

    success: {
        title: "Login Successful",
        message: "Welcome back to your account.",
        confirmText: "Continue"
    }

});
02

OTP Verification

await auth.confirmOtp({

    title: "Verify Your Account",

    message: "Enter the code sent to your email.",

    length: 6,

    resendDelay: 30,

    onVerify: async (code, { signal }) => {

        const response = await fetch("/api/verify-otp.php", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({ code }),
            signal
        });

        return response.json();
    }

});
03

Theme Customization

const auth = createAuthConfirm({

    theme: "dark",

    primaryColor: "#078b4d",

    confirmation: {
        timeout: 15000
    }

});

// Change theme whenever needed.
auth.setTheme("light");

// Update primary colour.
auth.setPrimaryColor("#2563eb");
