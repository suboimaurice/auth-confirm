JavaScript Frontend Integration 
import { createAuthConfirm } from "auth-confirm";

import "auth-confirm/style.css";

const auth = createAuthConfirm({

theme: "auto",

primaryColor: "#078b4d",

confirmation: {
    timeout: 15000
},

onConfirm: async (action, { signal }) => {

    const response = await fetch(
        `/api/${action}.php`,
        {
            method: "POST",
            signal
        }
    );

    const result = await response.json();

    return {
        success: response.ok && result.success === true,
        message: result.message
    };
}

});

// Open login confirmation.
await auth.confirmLogin();
PHP Backend Integration 
<?php

session_start();

header("Content-Type: application/json");

if ($_SERVER["REQUEST_METHOD"] !== "POST") {

http_response_code(405);

echo json_encode([
    "success" => false,
    "message" => "Method not allowed."
]);

exit;

}

$action = basename(
$_SERVER["SCRIPT_NAME"],
".php"
);

if ($action === "login") {

// Validate credentials against your database.

$_SESSION["user"] = [
    "id" => 1,
    "name" => "John Doe"
];

} elseif ($action === "logout") {

session_unset();
session_destroy();

} else {

http_response_code(400);

echo json_encode([
    "success" => false,
    "message" => "Invalid authentication action."
]);

exit;

}

echo json_encode([
"success" => true,
"message" => "Authentication completed."
]);

02

OTP Verification

const auth = createAuthConfirm({

otp: {
    length: 6,
    resendDelay: 30
},

onVerify: async (code, { signal }) => {

    const response = await fetch(
        "/api/verify-otp.php",
        {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({ code }),
            signal
        }
    );

    const result = await response.json();

    return result.success === true;
}

});

// Open OTP verification.
await auth.confirmOtp();
